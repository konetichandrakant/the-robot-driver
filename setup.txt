#====================================================
#  The Robot Driver - Docker Setup Guide
#  (Windows WSL example)
#====================================================
# Run these commands in your WSL terminal after installing:
#   - WSL
#   - Docker Desktop (with WSL integration enabled)
#   - An X server on Windows (e.g. VcXsrv / Xming)
#====================================================

#----------------------------------------
#  1. Display Environment Configuration
#----------------------------------------
# Configure X11 display forwarding for GUI applications in Docker
export DISPLAY=$(grep -m 1 nameserver /etc/resolv.conf | awk '{print $2}'):0

# Set OpenGL rendering mode for compatibility
export LIBGL_ALWAYS_INDIRECT=1

#----------------------------------------
#  2. Navigate to Project Directory
#----------------------------------------
# In my case:
cd /mnt/d/ck/projects/the-robot-driver

#----------------------------------------
#  3. Build Docker Image
#----------------------------------------
# (Optional) Clean up previous build cache
docker builder prune -f

# Build the Docker image with no cache for latest changes
docker build --no-cache -t the-robot-driver-app .

#----------------------------------------
#  4. Run Tests with Docker Compose
#----------------------------------------
# NOTE: docker-compose.yml is already configured with:
#   - service: automation_test      -> runs tests/test_automation_api.py
#   - service: login_task_test      -> runs tests/test_login_task.py (with GUI/X11)

# Run the automation API tests
docker compose run --rm automation_test

# Run the login task tests (GUI / browser tests)
docker compose run --rm login_task_test
